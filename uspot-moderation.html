⚙️ SQL для настройки таблиц

🔧 Шаг 1 — Запускайте хоть каждый раз (безопасно):

Добавляет колонки и обновляет RLS-политики. Не ломается при повторном запуске.

-- ── Колонки для masters (если ещё нет) ──────────────────────
ALTER TABLE masters ADD COLUMN IF NOT EXISTS rating numeric DEFAULT 5.0;
ALTER TABLE masters ADD COLUMN IF NOT EXISTS review_count integer DEFAULT 0;

-- ── Колонки для reviews (если ещё нет) ─────────────────────
ALTER TABLE reviews ADD COLUMN IF NOT EXISTS text text;
ALTER TABLE reviews ADD COLUMN IF NOT EXISTS status text DEFAULT 'pending';
ALTER TABLE reviews ADD COLUMN IF NOT EXISTS photo_urls jsonb DEFAULT '[]';
ALTER TABLE reviews ADD COLUMN IF NOT EXISTS client_name text;
ALTER TABLE reviews ADD COLUMN IF NOT EXISTS client_telegram_id text;
-- Backfill: старые отзывы без статуса → pending (попадут в очередь модерации)
UPDATE reviews SET status = 'pending' WHERE status IS NULL;

-- ── Колонки для bookings (если ещё нет) ─────────────────────
ALTER TABLE bookings ADD COLUMN IF NOT EXISTS client_notes text;
ALTER TABLE bookings ADD COLUMN IF NOT EXISTS client_photo_urls jsonb DEFAULT '[]';
ALTER TABLE bookings ADD COLUMN IF NOT EXISTS final_price numeric;
ALTER TABLE bookings ADD COLUMN IF NOT EXISTS master_note text;

-- ── Аватары (если ещё нет) ───────────────────────────────────
ALTER TABLE masters ADD COLUMN IF NOT EXISTS photo_url text;
ALTER TABLE clients ADD COLUMN IF NOT EXISTS photo_url text;

-- ── Даты акций (если ещё нет) ────────────────────────────────
ALTER TABLE promos ADD COLUMN IF NOT EXISTS valid_from date;

-- ── RLS: разрешить анон-ключу читать/писать всё (прототип) ───
DROP POLICY IF EXISTS "Anon schedule"  ON master_schedule;
CREATE POLICY "Anon schedule"  ON master_schedule FOR ALL USING (true) WITH CHECK (true);
DROP POLICY IF EXISTS "Anon bookings" ON bookings;
CREATE POLICY "Anon bookings" ON bookings  FOR ALL USING (true) WITH CHECK (true);
DROP POLICY IF EXISTS "Anon reviews"  ON reviews;
CREATE POLICY "Anon reviews"  ON reviews   FOR ALL USING (true) WITH CHECK (true);
DROP POLICY IF EXISTS "Anon portfolio" ON portfolio;
CREATE POLICY "Anon portfolio" ON portfolio FOR ALL USING (true) WITH CHECK (true);
DROP POLICY IF EXISTS "Anon clients"  ON clients;
CREATE POLICY "Anon clients"  ON clients   FOR ALL USING (true) WITH CHECK (true);

-- ── Таблица client_ratings (оценки клиентов от мастеров) ──────
CREATE TABLE IF NOT EXISTS client_ratings (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  master_id uuid,
  client_telegram_id text NOT NULL,
  booking_id uuid,
  rating integer NOT NULL CHECK (rating >= 1 AND rating <= 5),
  created_at timestamptz DEFAULT now()
);
ALTER TABLE client_ratings ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Anon client_ratings" ON client_ratings;
CREATE POLICY "Anon client_ratings" ON client_ratings FOR ALL USING (true) WITH CHECK (true);

-- ── Таблица master_blacklist ──────────────────────────────────
CREATE TABLE IF NOT EXISTS master_blacklist (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  master_id uuid NOT NULL,
  client_telegram_id text NOT NULL,
  client_name text,
  reason text,
  created_at timestamptz DEFAULT now(),
  UNIQUE(master_id, client_telegram_id)
);
ALTER TABLE master_blacklist ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Anon master_blacklist" ON master_blacklist;
CREATE POLICY "Anon master_blacklist" ON master_blacklist FOR ALL USING (true) WITH CHECK (true);

📸 Шаг 1б — Только если фото не загружаются (Storage RLS):

Запустите отдельным запросом — не вместе с Шагом 1.

-- Убедитесь что бакет "uspot-media" существует и Public
-- (Storage → New bucket → uspot-media → включить Public)

DROP POLICY IF EXISTS "Anon select uspot-media" ON storage.objects;
CREATE POLICY "Anon select uspot-media" ON storage.objects
  FOR SELECT USING (bucket_id = 'uspot-media');

DROP POLICY IF EXISTS "Anon insert uspot-media" ON storage.objects;
CREATE POLICY "Anon insert uspot-media" ON storage.objects
  FOR INSERT WITH CHECK (bucket_id = 'uspot-media');

DROP POLICY IF EXISTS "Anon update uspot-media" ON storage.objects;
CREATE POLICY "Anon update uspot-media" ON storage.objects
  FOR UPDATE USING (bucket_id = 'uspot-media');

DROP POLICY IF EXISTS "Anon delete uspot-media" ON storage.objects;
CREATE POLICY "Anon delete uspot-media" ON storage.objects
  FOR DELETE USING (bucket_id = 'uspot-media');

🆕 Шаг 2 — Только при первой установке (пропустите если уже запускали):

-- ── Таблица portfolio ────────────────────────────────────────
CREATE TABLE IF NOT EXISTS portfolio (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  master_id uuid REFERENCES masters(id) ON DELETE CASCADE,
  image_urls jsonb DEFAULT '[]',
  caption text,
  tag text,
  status text DEFAULT 'pending',
  created_at timestamptz DEFAULT now()
);
ALTER TABLE portfolio ENABLE ROW LEVEL SECURITY;

-- ── Уникальный индекс для расписания (один раз!) ─────────────

  ALTER TABLE master_schedule ADD CONSTRAINT master_schedule_master_day_unique
    UNIQUE (master_id, day_of_week);
EXCEPTION WHEN duplicate_table THEN NULL;
         WHEN others THEN NULL;
END $$;
